# Страница книжного клуба — план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Страница `/bookclub/` на xenitch.ru с формой регистрации, которая пишет заявку в гугл-таблицу челленджа через Apps Script и уводит на оплату ЮKassa.

**Architecture:** Статический сайт (build.py, Jinja2) получает новый шаблон с формой и inline-скриптом; таблица получает четвёртый файл Apps Script `club.gs` с `doPost`, развёрнутый как веб-приложение. Браузер шлёт JSON простым POST (text/plain), читает `{ok}` и делает редирект на ЮKassa.

**Tech Stack:** Python 3.9 + Jinja2 + pytest (сайт); Google Apps Script + jsc-тесты (таблица); vanilla JS на странице.

**Spec:** `site/_specs/2026-09-09-bookclub-page-design.md`

```js
/**
 * Регистрация в Органический книжный клуб — форма на xenitch.ru/bookclub/
 * шлёт заявку сюда, скрипт дописывает строку в лист «Книжный клуб».
 * Кладётся ЧЕТВЁРТЫМ файлом в тот же Apps Script проект, что и сборщики.
 *
 * УСТАНОВКА (один раз):
 * 1. Вставьте этот файл в проект скриптов таблицы (Расширения → Apps Script →
 *    «+» → Скрипт → назвать club).
 * 2. Развернуть → Новое развёртывание → тип «Веб-приложение»:
 *    выполнять «От моего имени», доступ «Все». Нажать «Начать развёртывание»,
 *    подтвердить доступ.
 * 3. Скопировать «URL веб-приложения» (…/exec) и вписать его в form_url
 *    страницы content/bookclub/index.md на сайте.
 * 4. При правках этого файла: Развернуть → Управление развёртываниями →
 *    карандаш → Версия: «Новая версия» → Развернуть. URL при этом не меняется.
 *
 * Оплата сверяется по почте вручную: колонка «Оплачено» заполняется в таблице.
 */

var CLUB_CONFIG = { SHEET: 'Книжный клуб' };
var CLUB_HEADERS = ['Дата', 'Имя', 'Телефон', 'Почта', 'Формат', 'Оплачено'];
var CLUB_FORMATS = ['Оффлайн, Кемерово, 28 сентября', 'Онлайн, Zoom, 29 сентября'];
var CLUB_MAX_LEN = 200;

// ==================== Чистые функции (тестируются локально) ====================

function clubField(obj, key) {
  var v = obj[key];
  if (v === undefined || v === null) return '';
  return String(v).replace(/\s+/g, ' ').trim().slice(0, CLUB_MAX_LEN);
}

// Разбор тела запроса. Возвращает {ok, row} | {ok, skip} | {ok:false, error}.
function parseRegistration(bodyText) {
  var data;
  try { data = JSON.parse(bodyText || ''); } catch (e) { data = null; }
  if (!data || typeof data !== 'object') return { ok: false, error: 'Неверный формат запроса' };
  if (clubField(data, 'website')) return { ok: true, skip: true };  // ловушка для ботов
  var name = clubField(data, 'name'), phone = clubField(data, 'phone');
  var email = clubField(data, 'email').toLowerCase(), format = clubField(data, 'format');
  if (!name) return { ok: false, error: 'Укажите имя' };
  if (!phone) return { ok: false, error: 'Укажите номер телефона' };
  if (!email) return { ok: false, error: 'Укажите эл. почту' };
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) return { ok: false, error: 'Проверьте эл. почту' };
  if (CLUB_FORMATS.indexOf(format) < 0) return { ok: false, error: 'Выберите формат участия' };
  return { ok: true, row: [name, phone, email, format] };
}

// ==================== Apps Script (веб-приложение) ====================

function clubSheet_() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(CLUB_CONFIG.SHEET);
  if (!sh) {
    sh = ss.insertSheet(CLUB_CONFIG.SHEET);
    sh.appendRow(CLUB_HEADERS);
    sh.setFrozenRows(1);
  }
  return sh;
}

function clubJson_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var parsed = parseRegistration(e && e.postData && e.postData.contents);
  if (!parsed.ok) return clubJson_(parsed);
  if (parsed.skip) return clubJson_({ ok: true });
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    clubSheet_().appendRow([new Date()].concat(parsed.row, ['']));
  } finally {
    lock.releaseLock();
  }
  return clubJson_({ ok: true });
}

function doGet() {
  return clubJson_({ ok: true, service: 'bookclub' });
}
```

<script>
(function () {
  var form = document.getElementById('reg-form');
  var error = document.getElementById('reg-error');
  var payUrl = {{ page.meta.pay_url | tojson }};
  var formUrl = {{ page.meta.form_url | tojson }};
  form.addEventListener('submit', function (ev) {
    ev.preventDefault();
    var button = form.querySelector('button');
    var data = {};
    new FormData(form).forEach(function (v, k) { data[k] = v; });

        location.href = payUrl;

      .catch(function () {
        error.hidden = false;
        button.disabled = false;
